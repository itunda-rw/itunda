Saronite gives Itunda a mini-app host and SDK so more experiences can live inside the same product without breaking the core journey.

import { AppRegistry } from 'react-native';
import PayBills from 'saronite-miniapp-pay-bills/app';
import WalletBalance from 'saronite-miniapp-wallet-balance/app';

AppRegistry.registerComponent('SaronitePayBills', () => PayBills);
AppRegistry.registerComponent('SaroniteWalletBalance', () => WalletBalance);
